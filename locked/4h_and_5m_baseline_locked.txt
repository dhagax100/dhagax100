// Measurement-only MFE/MAE from confirmed 1m OHLC AFTER the entry minute.
// The entry event is detected intraminute from that minute's high/low. Because
// 1m OHLC cannot tell whether the entry-minute low/high happened before or
// after the trigger, the whole entry minute is excluded from excursions.
// Measurement therefore starts at the first complete 1m bar with time > startT.
tradeExcursions1m(int startT, int endT, float entryP, float riskP, bool isBuy) =>
    float mfePx = 0.0
    float maePx = 0.0
    int mfeT = startT
    int maeT = startT
    int n1 = array.size(M1T)
    if n1 > 0 and endT > startT and riskP > 0
        int lo = 0
        int hi = n1
        while lo < hi
            int mid = int(math.floor((lo + hi) / 2))
            if array.get(M1T, mid) <= startT
                lo := mid + 1
            else
                hi := mid
        if lo < n1
            for q = lo to n1 - 1
                int tt = array.get(M1T, q)
                if tt > endT
                    break
                float fav = isBuy ? array.get(M1H, q) - entryP : entryP - array.get(M1L, q)
                float adv = isBuy ? entryP - array.get(M1L, q) : array.get(M1H, q) - entryP
                fav := math.max(fav, 0.0)
                adv := math.max(adv, 0.0)
                if fav > mfePx
                    mfePx := fav
                    mfeT := tt
                if adv > maePx
                    maePx := adv
                    maeT := tt
    [mfePx, maePx, mfeT, maeT]

first1mCross(int startT, int endT, float level, bool above) =>
    int rt = -1
    float ex = na
    int n1 = array.size(M1T)
    if n1 > 0 and endT > startT
        // M1T is chronological. Binary-search the first minute >= startT,
        // then inspect only the requested interval instead of rescanning the
        // entire 1m ledger on every structural query. Strategy semantics unchanged.
        int lo = 0
        int hi = n1
        while lo < hi
            int mid = int(math.floor((lo + hi) / 2))
            if array.get(M1T, mid) < startT
                lo := mid + 1
            else
                hi := mid
        int q = lo
        while q < n1 and rt == -1
            int tt = array.get(M1T, q)
            if tt >= endT
                break
            float hh = array.get(M1H, q)
            float ll = array.get(M1L, q)
            if (above and hh > level) or (not above and ll < level)
                rt := tt
                ex := above ? hh : ll
            q += 1
    [rt, ex]

// Unlike structural exceedance, a stop/target is reached on equality as well.
first1mTouch(int startT, int endT, float level, bool touchAbove) =>
    int rt = -1
    float ex = na
    int n1 = array.size(M1T)
    if n1 > 0 and endT > startT
        int lo = 0
        int hi = n1
        while lo < hi
            int mid = int(math.floor((lo + hi) / 2))
            if array.get(M1T, mid) < startT

            else
                hi := mid
        int q = lo
        while q < n1 and rt == -1
            int tt = array.get(M1T, q)

            if (touchAbove and hh >= level) or (not touchAbove and ll <= level)
                rt := tt
                ex := touchAbove ? hh : ll
            q += 1
    [rt, ex]

// Exact 1m event clock for NATIVE 5m swing confirmation.
// 5m SWH is confirmed when the confirming 5m candle first trades strictly
// below the previous 5m low. 5m SWL is the mirror: first strict trade above
// the previous 5m high. Event time is the 1m open; confirmation is +60s.
resolve5mSwingEvent1m(int confirmIdx, bool swingHighEvent) =>
    int rt = -1
    float ex = na
    if confirmIdx > 0 and confirmIdx < array.size(fT)
        int st = array.get(fT, confirmIdx)
        int en = st + 5 * 60 * 1000
        float lvl = swingHighEvent ? array.get(fL, confirmIdx - 1) : array.get(fH, confirmIdx - 1)
        [tt, xx] = first1mCross(st, en, lvl, not swingHighEvent)
        rt := tt
        ex := xx
    [rt, ex]

// Performance cache: resolve each native-5m swing's exact 1m event at most once.
// This removes repeated full M1 scans from drawAll() while preserving the event rule.
get5mSwingEventCached(int pos, bool swingHighEvent) =>
    int rt = -1
    if swingHighEvent
        int n = array.size(fSwHighConf)
        if pos >= 0 and pos < n
            while array.size(fSwHighEventT) < n
                array.push(fSwHighEventT, -2)
            int cached = array.get(fSwHighEventT, pos)
            if cached == -2
                int ci = array.get(fSwHighConf, pos)
                [tt, xx] = resolve5mSwingEvent1m(ci, true)
                cached := tt
                array.set(fSwHighEventT, pos, cached)
            rt := cached
    else
        int n = array.size(fSwLowConf)
        if pos >= 0 and pos < n
            while array.size(fSwLowEventT) < n
                array.push(fSwLowEventT, -2)
            int cached = array.get(fSwLowEventT, pos)
            if cached == -2
                int ci = array.get(fSwLowConf, pos)
                [tt, xx] = resolve5mSwingEvent1m(ci, false)
                cached := tt
                array.set(fSwLowEventT, pos, cached)
            rt := cached
    rt

// A waiting AOB becomes the IFOB on the matching MSS.  Do not promote a
// rejected, stranded, spent, or already-eligible zone.
promoteNativeWaitingAOB(int idx, bool bull) =>
    bool result = false
    if idx >= 0 and idx < array.size(obs)
        z = array.get(obs, idx)
        if z.state == 1 and z.origState == 1 and z.bullish == bull and z.eligibleK == -1 and not z.rejected
            z.state := 0
            z.origState := 0
            z.eligibleK := -1
            z.stopK := -1
            array.set(obs, idx, z)
            result := true
    result

// Is this exact candle already the origin of an existing zone (any state,
// any origin, alive or dead)? A fresh backward-scan can independently land
// on the same "best candle" a second time (e.g. once for the original
// AIFOB, again later for its promotion/fresh-scan fallback) -- this stops
// a second zone from being stacked on top of one that already claimed it.
candleClaimed(int cand, bool bull) =>
    found = false
    n = array.size(obs)
    for z = 0 to n - 1
        if z < 0 or z >= n
            continue
        if not found
            zz = array.get(obs, z)
            if zz.candle == cand and zz.bullish == bull
                found := true
    found

// A zone that was ever born as AIFOB keeps origState==4 forever (even after
// promotion to IFOB, or after dying). Used to skip the fresh-scan fallback
// when an AIFOB already exists for this exact setup -- one swing high and
// one swing low already existed before the break, so a second, separate
// IFOB in the same candle range would just be a duplicate.
existsAifobInRange(int lo, int hi, bool bull) =>
    found = false
    n = array.size(obs)
    for z = 0 to n - 1
        if z < 0 or z >= n
            continue
        if not found
            zz = array.get(obs, z)
            if zz.origState == 4 and zz.bullish == bull and zz.candle >= lo and zz.candle <= hi
                found := true
    found

tryBullAOB(int pReg, int aobSWHi, int newSwlI, float newSwlP, int k) =>

                    else if cX > array.get(C_, best2)

            if best2 != -1 and array.get(L_, best2) > newSwlP
                oB = array.get(O_, best2)
                cB = array.get(C_, best2)
                addOB(best2, math.min(oB,cB),math.max(oB,cB),false,k,1, array.get(H_, k-1), true)
                ai = array.size(obs)-1
                if ai >= 0 and k > 0

                    ep = array.get(H_,k-1)
                    if ep > aob.zt
                        aob.rejected := true
                        aob.eligibleK := -1
                        aob.stopK := -1
                    else
                        aob.eligibleK := k
                        if aob.eligibilityTime < 0
                            [et, ex] = resolveSwingConfirm1m(k, false)
                            aob.eligibilityTime := et >= 0 ? et : array.get(BT, k)
                            aob.eligibilityConfirmTime := aob.eligibilityTime + 60 * 1000
                            aob.eligibilityPrice := newSwlP
                            aob.eligibilityExtreme := ex
                    array.set(obs,ai,aob)

// --- AIFOB HUNT: Bullish (returns obs[] index or -1) ---
tryBullAIFOB(int pReg, bool pHaveSWH, int pSwhI, int pLastSWLi, int newSwlI, int k) =>
    result = -1
    // Exception: the confirming candle (k) can, in the same bar, ALSO break
    // below the swing low it just confirmed (newSwlI) -- an outside/engulfing
    // bar. That means we're not continuing bullish, we're reversing down in
    // the same breath (the "Second break: SWL" check right after this will
    // catch the regime flip). Don't create a bullish candidate on a premise
    // this same candle already broke.
    alreadyBrokeIt = array.get(L_, k) < array.get(L_, newSwlI)
    if pReg == 1 and pHaveSWH and pSwhI >= 0 and pLastSWLi >= 0 and not alreadyBrokeIt
        lo = math.max(0, math.min(math.min(pLastSWLi, newSwlI), pSwhI - 1))
        hi = math.max(math.max(pLastSWLi, newSwlI), pSwhI - 1)
        best = -1
        for x = lo to hi
            if x < 0
                continue
            cX = array.get(C_, x)
            oX = array.get(O_, x)
            okB = false
            if cX < oX
                if best == -1
                    okB := true
                else
                    okB := cX < array.get(C_, best)
            if okB
                best := x
        if best != -1 and not candleClaimed(best, true)
            oB = array.get(O_, best)
            cB = array.get(C_, best)
            addOB(best, math.min(oB, cB), math.max(oB, cB), true, k, 4, array.get(H_, k-1), true)
            result := array.size(obs) - 1
    result

// --- AIFOB HUNT: Bearish (returns obs[] index or -1) ---

    if pReg == 2 and pHaveSWL and pSwlI >= 0 and pLastSWHi >= 0 and not alreadyBrokeIt2
        lo = math.max(0, math.min(math.min(pLastSWHi, newSwhI), pSwlI - 1))
        hi = math.max(math.max(pLastSWHi, newSwhI), pSwlI - 1)

        for x = lo to hi
            if x < 0
                continue
            cX = array.get(C_, x)
            oX = array.get(O_, x)
            okB = false
            if cX > oX
                if best == -1
                    okB := true
                else
                    okB := cX > array.get(C_, best)
            if okB

        if best != -1 and not candleClaimed(best, false)

            addOB(best, math.min(oB, cB), math.max(oB, cB), false, k, 4, array.get(L_, k-1), false)

    result

pruneNativeTerminalZones() =>
    aoi = array.size(activeObs) - 1
    while aoi >= 0
        aidx = array.get(activeObs, aoi)
        azone = array.get(obs, aidx)
        // H4 OOB is terminal.  Once stranded, it is not a valid H4 POI and
        // must never receive a later eligibility, reaction, or impact.
        if azone.state == 2 or azone.state == 3 or azone.rejected
            array.remove(activeObs, aoi)
        aoi -= 1

// Process each native H4 bar, or one completed H4 bar on its final 5m bar.

processH4Bar = h4NativeContext or h4ClosedOnFiveMinute
if not skipBar and processH4Bar and bar_index != lastProcessedBI
    lastProcessedBI := bar_index

    engineHigh = h4NativeContext ? high : h4ProjectedHigh
    engineLow = h4NativeContext ? low : h4ProjectedLow
    engineClose = h4NativeContext ? close : h4ProjectedClose
    engineTime = h4NativeContext ? time : h4ProjectedTime
    array.push(O_, engineOpen)
    array.push(H_, engineHigh)
    array.push(L_, engineLow)
    array.push(C_, engineClose)
    array.push(BI, bar_index)
    array.push(BT, int(engineTime))

    n = array.size(O_)
    if n >= 2
        i = n - 1    // current bar index in our arrays

        // ========== SWING DETECTION ==========
        isBull   = array.get(C_, i) >= array.get(O_, i)
        brkH     = array.get(H_, i) > array.get(H_, i - 1)
        brkL     = array.get(L_, i) < array.get(L_, i - 1)
        dualAct  = brkH and brkL

        // prevDual: the previous candle confirmed BOTH a swing high and a
        // swing low in one bar (a dual-action/outside-bar candle). We keep
        // the (kind, swingIdx) of BOTH of those events -- not just a plain
        // yes/no flag -- so we can tell a genuine duplicate (this candle
        // re-confirming the EXACT same point again) apart from a totally
        // different, legitimate confirmation that just happens to land the
        // bar right after a dual-action candle (e.g. that same candle's
        // high separately qualifying as its own swing high later).
        prevDual = false
        pdKind1 = -1
        pdIdx1 = -1
        pdKind2 = -1
        pdIdx2 = -1
        eCnt = array.size(events)
        if eCnt >= 2
            ev1 = array.get(events, eCnt - 1)
            ev2 = array.get(events, eCnt - 2)
            if (ev1.kind != ev2.kind) and (ev1.confirmIdx == ev2.confirmIdx) and (ev1.confirmIdx == i - 1)
                prevDual := true
                pdKind1 := ev1.kind
                pdIdx1 := ev1.swingIdx
                pdKind2 := ev2.kind
                pdIdx2 := ev2.swingIdx

        evBeforeSwing = array.size(events)  // save for STEP 3

        if not isBull
            // BEARISH: High first, then Low
            if array.get(H_, i) > array.get(H_, peakIdx_)
                peakIdx_ := i
            if brkH
                lk = array.size(events) > 0 ? array.get(events, array.size(events) - 1).kind : -1
                // Would this re-confirm troughIdx_ as a swing LOW when the
                // dual-action candle already confirmed that exact (kind=1,
                // idx) pair? Only block that exact duplicate.
                blockDup = prevDual and not dualAct and ((pdKind1 == 1 and pdIdx1 == troughIdx_) or (pdKind2 == 1 and pdIdx2 == troughIdx_))
                if lk != 1 and not blockDup

            if array.get(L_, i) < array.get(L_, troughIdx_)

            if brkL

            if brkL

                                [et, ex] = resolveSwingConfirm1m(k, true)
                                obZ.eligibilityTime := et >= 0 ? et : array.get(BT, k)
                                obZ.eligibilityConfirmTime := obZ.eligibilityTime + 60 * 1000
                                obZ.eligibilityPrice := sEv.price
                                obZ.eligibilityExtreme := ex
                        array.set(obs, z, obZ)
            else
                if not swlCons
                    haveSWL  := true
                    swlPrice := sEv.price
                    swlIdx   := sEv.swingIdx
                lastSWLidx := sEv.swingIdx
                obSz2 = array.size(activeObs)
                for az2 = 0 to obSz2 - 1
                    if az2 < 0 or az2 >= obSz2
                        continue
                    z = array.get(activeObs, az2)
                    if z < 0 or z >= array.size(obs)
                        continue
                    obZ2 = array.get(obs, z)
                    // Bearish mirror of the same-bar plain IFOB rule.
                    bool sameBarPlainIfob = obZ2.state == 0 and obZ2.origState == 0 and k == obZ2.triggerK
                    if not obZ2.bullish and (obZ2.state == 0 or obZ2.state == 1 or obZ2.state == 4) and obZ2.eligibleK == -1 and not obZ2.rejected and (k > obZ2.triggerK or sameBarPlainIfob)
                        // Bearish AOB/AIFOB mirror: drawn box bottom, not wick.
                        sourceBoundaryOk = (obZ2.origState != 1 and obZ2.origState != 4) or obZ2.zb > sEv.price
                        ep = k > 0 ? array.get(H_,k-1) : na
                        if not sourceBoundaryOk or (not na(ep) and ep > obZ2.zt)
                            obZ2.rejected := true
                            obZ2.eligibleK := -1
                            obZ2.stopK := -1
                        else
                            obZ2.eligibleK := k
                            if obZ2.eligibilityTime < 0
                                [et, ex] = resolveSwingConfirm1m(k, false)
                                obZ2.eligibilityTime := et >= 0 ? et : array.get(BT, k)
                                obZ2.eligibilityConfirmTime := obZ2.eligibilityTime + 60 * 1000
                                obZ2.eligibilityPrice := sEv.price
                                obZ2.eligibilityExtreme := ex
                        array.set(obs, z, obZ2)
            ei += 1

        // STEP 3: OB lifecycle at this candle
        obTotal = array.size(activeObs)
        hK = array.get(H_, k)
        lK = array.get(L_, k)
        for az3 = 0 to obTotal - 1
            if az3 < 0 or az3 >= obTotal
                continue
            z = array.get(activeObs, az3)
            if z < 0 or z >= array.size(obs)
                continue
            obZ3 = array.get(obs, z)
            // An OOB has no further H4 lifecycle.  This guard also protects
            // the current pass before terminal zones are pruned below.
            if obZ3.state == 2 or obZ3.state == 3 or obZ3.rejected
                continue

            zb = obZ3.zb
            zt = obZ3.zt
            bull = obZ3.bullish

            // IMPACT
            if obZ3.eligibleK != -1 and k >= obZ3.eligibleK
                if hK >= zb and lK <= zt
                    // H4 impact lifecycle preserved; 5m resolver removed.
                    obZ3.preSpentState := obZ3.state
                    obZ3.state := 3
                    obZ3.stopK := k
                    if obZ3.impactTime < 0
                        int ist = obZ3.eligibilityTime >= 0 ? obZ3.eligibilityTime : array.get(BT, k)
                        int ien = array.get(BT, k) + 4 * 60 * 60 * 1000
                        float boundary = bull ? zt : zb
                        // Strict penetration: bullish Low < Top; bearish High > Bottom.
                        [it, ix] = first1mCross(ist, ien, boundary, not bull)
                        obZ3.impactTime := it >= 0 ? it : array.get(BT, k)
                        obZ3.impactConfirmTime := obZ3.impactTime + 60 * 1000
                        obZ3.impactPrice := boundary
                        obZ3.impactExtreme := ix
                    recordImpact(k)
                    array.set(obs, z, obZ3)
                    continue

            // STRANDING -> OOB
            if (obZ3.state == 0 or obZ3.state == 1 or obZ3.state == 4) and obZ3.eligibleK != -1
                isIFOB = obZ3.origState != 1
                // scan only events confirmed at THIS bar (they start at evBeforeSwing in worst case,
                // but are contiguous at end of events array for this bar)
                for e2 = evBeforeSwing to evTotal - 1
                    if e2 < 0 or e2 >= evTotal
                        continue
                    stEv = array.get(events, e2)
                    if stEv.confirmIdx != k
                        continue
                    stranded = false
                    if isIFOB
                        if bull and stEv.kind == 1 and stEv.price > zt
                            stranded := true
                        if not bull and stEv.kind == 0 and stEv.price < zb
                            stranded := true
                    else
                        // AOB stranding is away from its zone: bullish AOB
                        // needs a new swing low above its top; bearish AOB
                        // needs a new swing high below its bottom.
                        if bull and stEv.kind == 1 and stEv.price > zt
                            stranded := true
                        if not bull and stEv.kind == 0 and stEv.price < zb
                            stranded := true
                    if stranded
                        obZ3.state := 2
                        array.set(obs, z, obZ3)
                        break
        // Full history remains stored; only terminal entries leave the hot indexes.
        pruneNativeTerminalZones()

// ===================== PROACTIVE 1M IMPACT WATCHER ON 5M =====================
// H4 structure/creation/eligibility remains owned by the completed-H4 engine.
// Once an H4 OB is already eligible, impact no longer waits for that H4 candle
// to close. Every confirmed 5m bar contributes its confirmed 1m children above;
// the first strict 1m penetration is latched immediately. The stored impactTime
// is the single clock used by the box edge, impact line and audit table.
if h4FiveMinuteProjection and barstate.isconfirmed and not skipBar and array.size(M1T) > 0
    int liveN = array.size(activeObs)
    if liveN > 0
        for livePos = 0 to liveN - 1
            if livePos < 0 or livePos >= array.size(activeObs)
                continue
            int liveIdx = array.get(activeObs, livePos)
            if liveIdx < 0 or liveIdx >= array.size(obs)
                continue
            ObZone liveOb = array.get(obs, liveIdx)
            if liveOb.rejected or liveOb.state == 2 or liveOb.state == 3 or liveOb.eligibleK == -1 or liveOb.eligibilityTime < 0 or liveOb.impactTime >= 0
                continue

            // Search only confirmed 1m bars that belong to this just-closed 5m
            // candle, while never allowing an impact before eligibility.
            int scanStart = math.max(int(time), liveOb.eligibilityTime)
            int scanEnd = int(time_close)
            float boundary = liveOb.bullish ? liveOb.zt : liveOb.zb
            [liveImpactT, liveImpactX] = first1mCross(scanStart, scanEnd, boundary, not liveOb.bullish)
            if liveImpactT >= 0
                liveOb.preSpentState := liveOb.state
                liveOb.state := 3
                liveOb.impactTime := liveImpactT
                liveOb.impactConfirmTime := liveImpactT + 60 * 1000
                liveOb.impactPrice := boundary
                liveOb.impactExtreme := liveImpactX

                // stopK must point to the H4 candle CONTAINING the 1m impact,
                // never the next H4 processing candle.
                int containingH4 = -1
                int h4n = array.size(BT)
                if h4n > 0
                    for hk = h4n - 1 to 0
                        int hs = array.get(BT, hk)
                        if liveImpactT >= hs and liveImpactT < hs + 4 * 60 * 60 * 1000
                            containingH4 := hk
                            break
                liveOb.stopK := containingH4
                if containingH4 >= 0
                    recordImpact(containingH4)
                array.set(obs, liveIdx, liveOb)

// ===================== RECENT H4 OB DISPLAY FILTER =====================
// Additive verification filter only. It does not alter OB creation, state,
// eligibility, impact, rejection, or any other baseline lifecycle logic.
recentH4ObSelected(int obIndex, int fromTime) =>
    bool selected = false
    if obIndex >= 0 and obIndex < array.size(obs)
        ObZone candidate = array.get(obs, obIndex)
        if not candidate.rejected and candidate.triggerTime >= fromTime
            int newerValid = 0
            int totalObs = array.size(obs)
            if obIndex < totalObs - 1
                for ri = obIndex + 1 to totalObs - 1
                    ObZone newer = array.get(obs, ri)
                    if not newer.rejected and newer.triggerTime >= fromTime
                        newerValid += 1
            selected := newerValid < InpRecentH4OBCount
    selected

                                [breachT5, breachX5] = first1mCross(activeStartT, intervalEndT, invalidLevel, not buySetup)

                                bool gotEntry5 = entryT5 >= 0 and (breachT5 < 0 or entryT5 < breachT5) and (not closeInvalidInInterval5 or entryT5 < h4CloseInvalidT)
                                bool gotBreach5 = breachT5 >= 0 and (entryT5 < 0 or breachT5 <= entryT5) and (not closeInvalidInInterval5 or breachT5 < h4CloseInvalidT)
                                bool gotCloseInvalid5 = closeInvalidInInterval5 and not gotEntry5 and not gotBreach5
                                int lineStopT = intervalEndT
                                if gotEntry5
                                    lineStopT := entryT5
                                else if gotBreach5
                                    lineStopT := breachT5
                                else if gotCloseInvalid5
                                    lineStopT := h4CloseInvalidT

                                // Only ONE blue line is allowed per setup:
                                // - replaced candidates leave NO historical blue segment;
                                // - the current unresolved candidate is drawn to the current 5m edge;
                                // - the winning candidate becomes the official entry line and stops
                                //   at the exact 1m exceedance;
                                // - an H4 far-boundary breach removes the entry line completely.
                                int blueStartT = activeOriginT
                                bool rearmAfterSL = false
                                int rearmFromT = -1

                                // Per-trade ledger values must live in this enclosing scope.
                                // Pine does not allow us to create them only inside nested SL/TP
                                // branches and then use them afterward when writing the trade row.
                                float tradeSL5 = na
                                float tradeTP5 = na
                                string tradeResult5 = "ENTRY"
                                int tradeExitT5 = scanEndT

                                if gotEntry5
                                    // Trade numbering is 1-based and increments ONLY when an actual
                                    // blue candidate is exceeded and becomes an official entry.
                                    // This same event creates the chart trade and its table row.
                                    tradeAttemptNo += 1
                                    line bl = line.new(blueStartT, candLevel, entryT5, candLevel, xloc=xloc.bar_time, extend=extend.none, color=color.blue, width=2)
                                    array.push(drawnLines, bl)

                                    // Structural SL is locked from the safest relevant opposite
                                    // swing in the whole active candidate chain. BUY = lowest SWL;
                                    // SELL = highest SWH. No extra buffer is applied yet.
                                    float slRef5 = na
                                    if buySetup
                                        int nSafe = array.size(fSwLows)
                                        if nSafe > 0
                                            for sr = 0 to nSafe - 1
                                                int sri = array.get(fSwLows, sr)
                                                if sri >= 0 and sri < fn
                                                    int srt = array.get(fT, sri)
                                                    int srcEv = get5mSwingEventCached(sr, false)
                                                    int srcConfirmT = srcEv >= 0 ? srcEv + 60 * 1000 : (sr < array.size(fSwLowConf) ? array.get(fT, array.get(fSwLowConf, sr)) + 5 * 60 * 1000 : -1)
                                                    if srt >= riskStartT and srt <= entryT5 and srcConfirmT >= 0 and srcConfirmT <= entryT5
                                                        float srp = array.get(fL, sri)
                                                        slRef5 := na(slRef5) ? srp : math.min(slRef5, srp)
                                    else
                                        int nSafe = array.size(fSwHighs)
                                        if nSafe > 0
                                            for sr = 0 to nSafe - 1
                                                int sri = array.get(fSwHighs, sr)
                                                if sri >= 0 and sri < fn
                                                    int srt = array.get(fT, sri)
                                                    int srcEv = get5mSwingEventCached(sr, true)
                                                    int srcConfirmT = srcEv >= 0 ? srcEv + 60 * 1000 : (sr < array.size(fSwHighConf) ? array.get(fT, array.get(fSwHighConf, sr)) + 5 * 60 * 1000 : -1)
                                                    if srt >= riskStartT and srt <= entryT5 and srcConfirmT >= 0 and srcConfirmT <= entryT5
                                                        float srp = array.get(fH, sri)
                                                        slRef5 := na(slRef5) ? srp : math.max(slRef5, srp)

                                    if not na(slRef5)
                                        array.set(fiveSLPriceByOb, oi5, slRef5)

                                        // Locked 3R target. Risk is the absolute Entry-to-SL distance.
                                        // BUY TP = Entry + 3R; SELL TP = Entry - 3R.
                                        float risk5 = math.abs(candLevel - slRef5)
                                        float tp5 = buySetup ? candLevel + 3.0 * risk5 : candLevel - 3.0 * risk5
                                        array.set(fiveTPPriceByOb, oi5, tp5)
                                        tradeSL5 := slRef5
                                        tradeTP5 := tp5

                                        // ADDITIVE H4-STRUCTURE BREAK-EVEN LAYER.
                                        // BUY: while this trade is open, the first native H4 SWL whose
                                        // swing PRICE is inside this H4 OB activates BE at that swing's
                                        // exact 1m confirmation time. SELL mirrors with an H4 SWH.
                                        // The swing may originate before/after entry; only its confirmation
                                        // must be strictly after entry. BE price is exactly the entry level.
                                        int beActivateT5 = -1
                                        int h4EvN5 = array.size(events)
                                        if h4EvN5 > 0
                                            for hev5 = 0 to h4EvN5 - 1
                                                if beActivateT5 == -1
                                                    SwEv hse5 = array.get(events, hev5)
                                                    bool wantedH4Swing5 = buySetup ? hse5.kind == 1 : hse5.kind == 0
                                                    bool swingInsideOb5 = hse5.price >= oz5.zb and hse5.price <= oz5.zt
                                                    if wantedH4Swing5 and swingInsideOb5 and hse5.confirmIdx >= 0 and hse5.confirmIdx < array.size(BT)
                                                        [h4BeEventT5, h4BeExtreme5] = resolveSwingConfirm1m(hse5.confirmIdx, hse5.kind == 0)
                                                        int h4BeConfirmT5 = h4BeEventT5 >= 0 ? h4BeEventT5 + 60 * 1000 : array.get(BT, hse5.confirmIdx) + 4 * 60 * 60 * 1000
                                                        if h4BeConfirmT5 > entryT5 and h4BeConfirmT5 <= scanEndT
                                                            beActivateT5 := h4BeConfirmT5

                                        // First establish whether the original SL or TP ends the trade BEFORE BE.
                                        int preBeEndT5 = beActivateT5 >= 0 ? beActivateT5 : scanEndT
                                        [slHitT5, slHitX5] = first1mTouch(entryT5, preBeEndT5, slRef5, not buySetup)
                                        [tpPreBeT5, tpPreBeX5] = first1mTouch(entryT5, preBeEndT5, tp5, buySetup)
                                        bool preSameMinute5 = slHitT5 >= 0 and tpPreBeT5 >= 0 and slHitT5 == tpPreBeT5
                                        bool slWins5 = slHitT5 >= 0 and not preSameMinute5 and (tpPreBeT5 < 0 or slHitT5 < tpPreBeT5)
                                        bool tpWinsPreBe5 = tpPreBeT5 >= 0 and not preSameMinute5 and (slHitT5 < 0 or tpPreBeT5 < slHitT5)

                                        if slWins5
                                            line sln = line.new(blueStartT, slRef5, slHitT5, slRef5, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                            array.push(drawnLines, sln)
                                            array.set(fiveStageByOb, oi5, "SL")
                                            tradeResult5 := "SL"
                                            tradeExitT5 := slHitT5
                                            rearmAfterSL := true
                                            rearmFromT := slHitT5
                                        else if tpWinsPreBe5
                                            line tpn = line.new(blueStartT, tp5, tpPreBeT5, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                            array.push(drawnLines, tpn)
                                            array.set(fiveStageByOb, oi5, "TP")
                                            tradeResult5 := "TP"
                                            tradeExitT5 := tpPreBeT5
                                        else if preSameMinute5
                                            line sln = line.new(blueStartT, slRef5, slHitT5, slRef5, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                            line tpn = line.new(blueStartT, tp5, tpPreBeT5, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                            array.push(drawnLines, sln)
                                            array.push(drawnLines, tpn)
                                            array.set(fiveStageByOb, oi5, "AMBIGUOUS")
                                            tradeResult5 := "AMBIGUOUS"
                                            tradeExitT5 := slHitT5
                                        else if beActivateT5 >= 0
                                            // BE activated before either original exit. Original structural SL
                                            // is effective only up to BE activation; from then on stop = Entry.
                                            line oldSln = line.new(blueStartT, slRef5, beActivateT5, slRef5, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                            array.push(drawnLines, oldSln)
                                            [beHitT5, beHitX5] = first1mTouch(beActivateT5, scanEndT, candLevel, not buySetup)
                                            [tpPostBeT5, tpPostBeX5] = first1mTouch(beActivateT5, scanEndT, tp5, buySetup)
                                            bool beTpSame5 = beHitT5 >= 0 and tpPostBeT5 >= 0 and beHitT5 == tpPostBeT5
                                            bool beWins5 = beHitT5 >= 0 and not beTpSame5 and (tpPostBeT5 < 0 or beHitT5 < tpPostBeT5)
                                            bool tpWinsPostBe5 = tpPostBeT5 >= 0 and not beTpSame5 and (beHitT5 < 0 or tpPostBeT5 < beHitT5)
                                            if beWins5
                                                line beln = line.new(beActivateT5, candLevel, beHitT5, candLevel, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                                array.push(drawnLines, beln)
                                                array.set(fiveStageByOb, oi5, "BE")
                                                tradeResult5 := "BE"
                                                tradeExitT5 := beHitT5
                                                // BE is not an SL: no post-SL re-entry is armed here.
                                            else if tpWinsPostBe5
                                                line tpn = line.new(blueStartT, tp5, tpPostBeT5, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                                array.push(drawnLines, tpn)
                                                array.set(fiveStageByOb, oi5, "TP")
                                                tradeResult5 := "TP"
                                                tradeExitT5 := tpPostBeT5
                                            else if beTpSame5
                                                line beln = line.new(beActivateT5, candLevel, beHitT5, candLevel, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                                line tpn = line.new(blueStartT, tp5, tpPostBeT5, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                                array.push(drawnLines, beln)
                                                array.push(drawnLines, tpn)
                                                array.set(fiveStageByOb, oi5, "AMBIGUOUS")
                                                tradeResult5 := "AMBIGUOUS"
                                                tradeExitT5 := beHitT5
                                            else
                                                line beln = line.new(beActivateT5, candLevel, scanEndT, candLevel, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                                line tpn = line.new(blueStartT, tp5, scanEndT, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                                array.push(drawnLines, beln)
                                                array.push(drawnLines, tpn)
                                                array.set(fiveStageByOb, oi5, "BE ACTIVE")
                                                tradeResult5 := "ACTIVE"
                                        else
                                            // No BE trigger yet: preserve the locked active-trade visuals.
                                            line sln = line.new(blueStartT, slRef5, scanEndT, slRef5, xloc=xloc.bar_time, extend=extend.none, color=color.red, width=2)
                                            line tpn = line.new(blueStartT, tp5, scanEndT, tp5, xloc=xloc.bar_time, extend=extend.none, color=color.green, width=2)
                                            array.push(drawnLines, sln)
                                            array.push(drawnLines, tpn)
                                            array.set(fiveStageByOb, oi5, "TRADE ACTIVE")
                                            tradeResult5 := "ACTIVE"
                                    else
                                        array.set(fiveStageByOb, oi5, "ENTRY")

                                    // Measurement-only analytics: use original structural risk as 1R.
                                    float tradeMfePips5 = na
                                    float tradeMfeR5 = na
                                    int tradeMfeT5 = -1
                                    float tradeMaePips5 = na
                                    float tradeMaeR5 = na
                                    int tradeMaeT5 = -1
                                    if tradeSL5 == tradeSL5 and tradeExitT5 >= entryT5
                                        float initialRisk5 = math.abs(candLevel - tradeSL5)
                                        if initialRisk5 > 0
                                            [mfePx5, maePx5, mfeWhen5, maeWhen5] = tradeExcursions1m(entryT5, tradeExitT5, candLevel, initialRisk5, buySetup)
                                            float pipSizeM5 = syminfo.type == "forex" ? syminfo.mintick * 10.0 : syminfo.mintick
                                            tradeMfePips5 := pipSizeM5 > 0 ? mfePx5 / pipSizeM5 : na
                                            tradeMaePips5 := pipSizeM5 > 0 ? maePx5 / pipSizeM5 : na
                                            tradeMfeR5 := mfePx5 / initialRisk5
                                            tradeMaeR5 := maePx5 / initialRisk5
                                            tradeMfeT5 := mfeWhen5
                                            tradeMaeT5 := maeWhen5

                                    // Preserve every trade attempt separately for the table.
                                    array.push(fiveTradeOb, oi5)
                                    array.push(fiveTradeNo, tradeAttemptNo)
                                    array.push(fiveTradeEntryT, entryT5)
                                    array.push(fiveTradeEntryP, candLevel)
                                    array.push(fiveTradeSL, tradeSL5)
                                    array.push(fiveTradeTP, tradeTP5)
                                    array.push(fiveTradeResult, tradeResult5)
                                    array.push(fiveTradeMfePips, tradeMfePips5)
                                    array.push(fiveTradeMfeR, tradeMfeR5)
                                    array.push(fiveTradeMfeT, tradeMfeT5)
                                    array.push(fiveTradeMaePips, tradeMaePips5)
                                    array.push(fiveTradeMaeR, tradeMaeR5)
                                    array.push(fiveTradeMaeT, tradeMaeT5)
                                    array.push(fiveTradeExitT, tradeExitT5)
                                    array.set(fiveEntryTimeByOb, oi5, entryT5)
                                    array.set(fiveEntryPriceByOb, oi5, candLevel)
                                else if gotBreach5
                                    // Intentionally draw NO blue line after invalidation.
                                    array.set(fiveStageByOb, oi5, "H4 OB BREACHED")
                                else if gotCloseInvalid5
                                    // Same-TF close invalidation terminates all NEW 5m hunting.
                                    // No unresolved blue candidate survives the H4 close.
                                    array.set(fiveStageByOb, oi5, "H4 CLOSE INVALID")
                                else if nextCandOrigin == -1
                                    line bl = line.new(blueStartT, candLevel, huntEndT, candLevel, xloc=xloc.bar_time, extend=extend.none, color=color.blue, width=2)
                                    array.push(drawnLines, bl)
                                    array.set(fiveStageByOb, oi5, "CANDIDATE")

                                if rearmAfterSL
                                    // Post-SL re-entry mirrors the ORIGINAL entry sequence.
                                    // BUY: first SWL whose ORIGIN may be the SL-hit 5m candle,
                                    //      confirmed strictly after the exact SL hit -> immediately
                                    //      preceding confirmed SWH becomes the next candidate.
                                    // SELL mirrors: first SWH -> immediately preceding SWL.
                                    int reRestOrigin = -1
                                    int reRestConfirm = -1
                                    int reRestConfirmT = -1

                                    if buySetup
                                        int rrn = array.size(fSwLows)
                                        if rrn > 0
                                            for rrp = 0 to rrn - 1
                                                if reRestOrigin == -1
                                                    int rri = array.get(fSwLows, rrp)
                                                    int rrc = rrp < array.size(fSwLowConf) ? array.get(fSwLowConf, rrp) : -1
                                                    if rri >= 0 and rri < fn and rrc >= 0 and rrc < fn
                                                        int rrOT = array.get(fT, rri)
                                                        int rrEv = get5mSwingEventCached(rrp, false)
                                                        int rrCT = rrEv >= 0 ? rrEv + 60 * 1000 : array.get(fT, rrc) + 5 * 60 * 1000
                                                        // The origin may be the 5m candle CONTAINING the SL hit.
                                                        bool originAtOrAfterSLBar = rrOT <= rearmFromT and rearmFromT < rrOT + 5 * 60 * 1000 or rrOT > rearmFromT
                                                        if originAtOrAfterSLBar and rrCT > rearmFromT
                                                            reRestOrigin := rri
                                                            reRestConfirm := rrc
                                                            reRestConfirmT := rrCT
                                    else
                                        int rrn = array.size(fSwHighs)
                                        if rrn > 0
                                            for rrp = 0 to rrn - 1
                                                if reRestOrigin == -1
                                                    int rri = array.get(fSwHighs, rrp)
                                                    int rrc = rrp < array.size(fSwHighConf) ? array.get(fSwHighConf, rrp) : -1
                                                    if rri >= 0 and rri < fn and rrc >= 0 and rrc < fn
                                                        int rrOT = array.get(fT, rri)
                                                        int rrEv = get5mSwingEventCached(rrp, true)
                                                        int rrCT = rrEv >= 0 ? rrEv + 60 * 1000 : array.get(fT, rrc) + 5 * 60 * 1000
                                                        bool originAtOrAfterSLBar = rrOT <= rearmFromT and rearmFromT < rrOT + 5 * 60 * 1000 or rrOT > rearmFromT
                                                        if originAtOrAfterSLBar and rrCT > rearmFromT
                                                            reRestOrigin := rri
                                                            reRestConfirm := rrc
                                                            reRestConfirmT := rrCT

                                    int reCandOrigin = -1
                                    int reCandConfirm = -1
                                    int reCandConfirmT = -1
                                    float reCandLevel = na

                                    if reRestOrigin != -1
                                        if buySetup
                                            int rn = array.size(fSwHighs)
                                            if rn > 0
                                                // Immediate preceding SWH, exactly like Candidate #1.
                                                for rp2 = rn - 1 to 0
                                                    int ri2 = array.get(fSwHighs, rp2)
                                                    int rc2 = rp2 < array.size(fSwHighConf) ? array.get(fSwHighConf, rp2) : -1
                                                    if ri2 < reRestOrigin and rc2 >= 0 and rc2 <= reRestConfirm
                                                        reCandOrigin := ri2
                                                        reCandConfirm := rc2
                                                        // Candidate becomes actionable only when the NEW rest is confirmed.
                                                        reCandConfirmT := reRestConfirmT
                                                        reCandLevel := array.get(fH, ri2)
                                                        break
                                        else
                                            int rn = array.size(fSwLows)
                                            if rn > 0
                                                // Immediate preceding SWL, exactly like Candidate #1.
                                                for rp2 = rn - 1 to 0
                                                    int ri2 = array.get(fSwLows, rp2)
                                                    int rc2 = rp2 < array.size(fSwLowConf) ? array.get(fSwLowConf, rp2) : -1
                                                    if ri2 < reRestOrigin and rc2 >= 0 and rc2 <= reRestConfirm
                                                        reCandOrigin := ri2
                                                        reCandConfirm := rc2
                                                        reCandConfirmT := reRestConfirmT
                                                        reCandLevel := array.get(fL, ri2)
                                                        break

                                    // H4 far-boundary remains authoritative while waiting for the new rest/candidate.
                                    int reRaceEndT = reRestConfirmT >= 0 ? math.min(reRestConfirmT, huntEndT) : huntEndT
                                    [reBreachT, reBreachX] = first1mCross(rearmFromT, reRaceEndT, invalidLevel, not buySetup)
                                    if reBreachT >= 0
                                        array.set(fiveStageByOb, oi5, "H4 OB BREACHED")
                                        chainDone := true
                                    else if h4CloseInvalidT >= 0 and h4CloseInvalidT <= reRaceEndT
                                        array.set(fiveStageByOb, oi5, "H4 CLOSE INVALID")
                                        chainDone := true
                                    else if reRestOrigin == -1
                                        array.set(fiveStageByOb, oi5, "WAIT RE-REST")
                                        chainDone := true
                                    else if reCandOrigin == -1
                                        array.set(fiveStageByOb, oi5, "WAIT RE-CANDIDATE")
                                        chainDone := true
                                    else
                                        // Explicitly show the new post-SL resting swing that re-arms the hunt.
                                        if buySetup
                                            label rrLab = label.new(array.get(fT, reRestOrigin), array.get(fL, reRestOrigin) - swLowGapOffset, "▼", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.black, size=size.small)
                                            array.push(drawnLabels, rrLab)
                                        else
                                            label rrLab = label.new(array.get(fT, reRestOrigin), array.get(fH, reRestOrigin), "▲", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.blue, size=size.small)
                                            array.push(drawnLabels, rrLab)

                                        // Show the immediately preceding opposite swing as the newly armed candidate.
                                        if buySetup
                                            label reLab = label.new(array.get(fT, reCandOrigin), array.get(fH, reCandOrigin), "▲", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.blue, size=size.small)
                                            array.push(drawnLabels, reLab)
                                        else
                                            label reLab = label.new(array.get(fT, reCandOrigin), array.get(fL, reCandOrigin) - swLowGapOffset, "▼", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.black, size=size.small)
                                            array.push(drawnLabels, reLab)

                                        candOriginIdx := reCandOrigin
                                        candConfirmIdx := reCandConfirm
                                        candLevel := reCandLevel
                                        activeOriginT := array.get(fT, reCandOrigin)
                                        activeStartT := reCandConfirmT
                                        // Keep the locked cumulative structural SL pool from original H4 impact.
                                        riskStartT := impact5mOpenT
                                        // Continue the normal candidate-chain search AFTER the new rest.
                                        searchAfterOrigin := reRestOrigin
                                        array.set(fiveStageByOb, oi5, "RE-ENTRY HUNT")
                                        chainDone := false
                                else if gotEntry5 or gotBreach5 or gotCloseInvalid5 or nextCandOrigin == -1
                                    chainDone := true
                                else
                                    // Before promoting the next candidate, expose ALL
                                    // opposite swings whose origins occurred between the
                                    // old and new candidates. These are the structural
                                    // points relevant to the eventual safest SL.
                                    if buySetup
                                        int nslRel = array.size(fSwLows)
                                        if nslRel > 0
                                            for rp = 0 to nslRel - 1
                                                int ri = array.get(fSwLows, rp)
                                                if ri > searchAfterOrigin and ri < nextCandOrigin
                                                    label rl = label.new(array.get(fT, ri), array.get(fL, ri) - swLowGapOffset, "▼", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.black, size=size.small)
                                                    array.push(drawnLabels, rl)
                                        label ncLab = label.new(array.get(fT, nextCandOrigin), array.get(fH, nextCandOrigin), "▲", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.blue, size=size.small)
                                        array.push(drawnLabels, ncLab)
                                    else
                                        int nshRel = array.size(fSwHighs)
                                        if nshRel > 0
                                            for rp = 0 to nshRel - 1
                                                int ri = array.get(fSwHighs, rp)
                                                if ri > searchAfterOrigin and ri < nextCandOrigin
                                                    label rh = label.new(array.get(fT, ri), array.get(fH, ri), "▲", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.blue, size=size.small)
                                                    array.push(drawnLabels, rh)
                                        label ncLab = label.new(array.get(fT, nextCandOrigin), array.get(fL, nextCandOrigin) - swLowGapOffset, "▼", xloc=xloc.bar_time, yloc=yloc.price, style=label.style_none, textcolor=color.black, size=size.small)
                                        array.push(drawnLabels, ncLab)

                                    candOriginIdx := nextCandOrigin
                                    candConfirmIdx := nextCandConfirm
                                    candLevel := nextCandLevel
                                    activeOriginT := array.get(fT, nextCandOrigin)
                                    activeStartT := nextCandConfirmT
                                    searchAfterOrigin := nextCandOrigin

    // --- OB zones: hollow boxes ---
    // IFOB bull=blue, IFOB bear=black, AOB=green, OOB=red, AIFOB=orange
    // SPENT uses preSpentState color
    obSzD = array.size(obs)
    extT = time + 50 * 4 * 60 * 60 * 1000

    for z = 0 to obSzD - 1
        if z < 0 or z >= obSzD
            continue
        obD = array.get(obs, z)
        if obD.rejected
            continue
        if not recentH4ObSelected(z, dispFrom)
            continue
        idx = obD.candle
        if idx < 0 or idx >= nArr
            continue
        // Focus filter
        if InpFocusOB > 0
            if obSzD - z != InpFocusOB
                continue
        else
            bt = array.get(BT, idx)
            if bt < dispFrom
                continue

        leftT = array.get(BT, idx)
        rightT = extT
        if obD.impactTime >= 0
            // Exact visual endpoint: the OB box stops at the stored 1m impact event.
            rightT := obD.impactTime
        else if obD.stopK != -1 and obD.stopK < nArr
            // Non-impact terminal states may still use their native H4 stop index.
            rightT := array.get(BT, obD.stopK)

        // Color
        dSt = obD.state == 3 ? obD.preSpentState : obD.state
        // AIFOB display RE-ENABLED (matches ICT_OB_Diagnostic.pine) -- we're
        // actively verifying its creation/promotion logic now, need to see
        // it drawn. Orange = still-pending AIFOB (dSt==4, not yet promoted).
        // OOB display disabled by request on 5m/1h/4h -- never used at those
        // timeframes per the trading cascade (docs/trading_logic.md), so hide
        // the clutter there. Stranding detection/lifecycle above (a zone can
        // still go OOB and later flip to SPENT on touch) is untouched, and
        // OOB still draws normally on Daily/Weekly/Monthly/any other TF.
        if dSt == 2 and hideOOBHere
            continue
        col = color.blue
        if dSt == 0
            col := obD.bullish ? color.blue : color.black
        else if dSt == 1
            col := color.green
        else if dSt == 2
            col := color.red
        else if dSt == 4
            col := color.orange
        else
            col := obD.bullish ? color.blue : color.black

        // DISPLAY-ONLY H4 endpoint mapping. Keep the stored impactTime exact for
        // lifecycle/table/5m drawings. On a native H4 chart, an intrabar 1m time
        // has no native H4 x-coordinate, so map ONLY the box's visual right edge
        // to the OPEN of the H4 candle that contains the exact impact. This is the
        // same containing-candle rule already locked for the purple impact line.
        int visualRightT = rightT
        if h4NativeContext and obD.impactTime >= 0
            int btNBox = array.size(BT)
            int containingBox = -1
            if btNBox > 0
                for bbi = 0 to btNBox - 1
                    int bbst = array.get(BT, bbi)
                    int bben = bbi + 1 < btNBox ? array.get(BT, bbi + 1) : bbst + 4 * 60 * 60 * 1000
                    if obD.impactTime >= bbst and obD.impactTime < bben
                        containingBox := bbi
                        break
            if containingBox >= 0
                visualRightT := array.get(BT, containingBox)

        bx = box.new(leftT, obD.zt, visualRightT, obD.zb, border_color=col, border_width=1, bgcolor=na, xloc=xloc.bar_time)
        array.push(drawnBoxes, bx)

h4AuditFrom = useDisplayFrom ? InpDisplayFrom : last_bar_time - InpMonthsBack * 30 * 24 * 60 * 60 * 1000

drawH4ImpactLines() =>
    if h4EngineContext
        int count = 0
        int obCount = array.size(obs)
        if obCount > 0
            for ai = 0 to obCount - 1
                ObZone az = array.get(obs, ai)
                int originT = az.candle >= 0 and az.candle < array.size(BT) ? array.get(BT, az.candle) : na
                // VISUAL CLOCK LOCK: impact drawings consume ONLY the stored exact
                // 1m impact timestamp. stopK / parent-H4 indexing must never move
                // the line to an H4 candle boundary or to the next H4 candle.
                bool hasExactImpact = az.impactTime >= 0
                if not az.rejected and recentH4ObSelected(ai, h4AuditFrom) and not na(originT) and originT >= h4AuditFrom and hasExactImpact
                    count += 1
                    if InpFocusH4AuditID == 0 or count == InpFocusH4AuditID
                        int impactT = az.impactTime
                        // DISPLAY-ONLY MAPPING:
                        // - On 5m keep the exact stored 1m impact timestamp unchanged.
                        // - On native H4, an intrabar timestamp (e.g. 02:28) has no native
                        //   H4 x-coordinate. Anchor the line to the OPEN of the H4 candle
                        //   that CONTAINS that exact impact, never to the following candle.
                        // The ledger/table and OB impactTime remain exact 1m values.
                        int visualImpactT = impactT
                        if h4NativeContext
                            int btN = array.size(BT)
                            int containing = -1
                            if btN > 0
                                for bi = 0 to btN - 1
                                    int bst = array.get(BT, bi)
                                    int ben = bi + 1 < btN ? array.get(BT, bi + 1) : bst + 4 * 60 * 60 * 1000
                                    if impactT >= bst and impactT < ben
                                        containing := bi
                                        break
                            if containing >= 0
                                visualImpactT := array.get(BT, containing)
                        float anchor = az.impactPrice == az.impactPrice ? az.impactPrice : (az.bullish ? az.zt : az.zb)
                        float pad = math.max(syminfo.mintick, math.abs(az.zt - az.zb))
                        ln = line.new(visualImpactT, anchor - pad, visualImpactT, anchor + pad, xloc=xloc.bar_time, extend=extend.both, color=color.purple, width=2)
                        array.push(drawnLines, ln)

fmtEvent(int t, float p) =>
    t >= 0 ? str.format_time(t, "MM-dd HH:mm") + "\n@ " + str.tostring(p, format.mintick) : "—"

drawCompactH4Table() =>
    // Dedicated H4 table. It is physically separated from the 5m table so an
    // inactive table object can never cover it on the native H4 chart.
    table.clear(h4CompactTable, 0, 0, 7, 20)
    if InpShowH4Table and h4NativeContext
        color hb = color.new(color.purple, 10)
        table.cell(h4CompactTable, 0, 0, "H4 OB", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 1, 0, "Type", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 2, 0, "Side", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 3, 0, "Zone", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 4, 0, "Trigger", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 5, 0, "Eligible", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 6, 0, "Impact", text_color=color.white, bgcolor=hb)
        table.cell(h4CompactTable, 7, 0, "Status", text_color=color.white, bgcolor=hb)
        int shown = 0
        int obCount = array.size(obs)
        if obCount > 0
            for ai = obCount - 1 to 0
                if shown >= InpH4TableRows
                    break
                ObZone az = array.get(obs, ai)
                if recentH4ObSelected(ai, h4AuditFrom) and az.triggerTime >= h4AuditFrom
                    string os = az.origState == 0 ? "IFOB" : az.origState == 1 ? "AOB" : az.origState == 4 ? "AIFOB" : "OOB"
                    int ds = az.state == 3 ? az.preSpentState : az.state
                    string cs = ds == 0 ? "IFOB" : ds == 1 ? "AOB" : ds == 2 ? "OOB" : ds == 4 ? "AIFOB" : "SPENT"
                    string typ = os == cs ? os : os + "→" + cs
                    string side = az.bullish ? "BUY" : "SELL"
                    string zone = str.tostring(az.zb, format.mintick) + "–" + str.tostring(az.zt, format.mintick)
                    string trig = fmtEvent(az.triggerTime, az.triggerPrice)
                    string elig = fmtEvent(az.eligibilityTime, az.eligibilityPrice)
                    string imp = fmtEvent(az.impactTime, az.impactPrice)
                    bool timingBad = (az.triggerTime >= 0 and az.triggerExtreme != az.triggerExtreme) or (az.impactTime >= 0 and az.impactExtreme != az.impactExtreme)
                    string status = az.rejected ? "REJECTED" : az.state == 3 ? "IMPACTED" : cs
                    if timingBad
                        status += " !TIME"
                    int row = shown + 1
                    table.cell(h4CompactTable, 0, row, "#" + str.tostring(ai + 1))
                    table.cell(h4CompactTable, 1, row, typ)
                    table.cell(h4CompactTable, 2, row, side, text_color=az.bullish ? color.blue : color.black)
                    table.cell(h4CompactTable, 3, row, zone)
                    table.cell(h4CompactTable, 4, row, trig)
                    table.cell(h4CompactTable, 5, row, elig)
                    table.cell(h4CompactTable, 6, row, imp)
                    table.cell(h4CompactTable, 7, row, status)
                    shown += 1
        if shown == 0
            table.cell(h4CompactTable, 0, 1, "No H4 OBs in range", text_color=color.gray)

draw5mEntryTable() =>
    table.clear(fiveMinuteEntryTable, 0, 0, 10, 20)
    if InpShow5mTable and h4FiveMinuteProjection
        color hb = color.new(color.purple, 10)
        table.cell(fiveMinuteEntryTable, 0, 0, "Setup", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 1, 0, "Side", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 2, 0, "H4 POI", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 3, 0, "H4 Impact", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 4, 0, "5m Stage", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 5, 0, "Entry", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 6, 0, "SL", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 7, 0, "TP", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 8, 0, "Result", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 9, 0, "MFE", text_color=color.white, bgcolor=hb)
        table.cell(fiveMinuteEntryTable, 10, 0, "MAE", text_color=color.white, bgcolor=hb)
        int shownSetups = 0
        int outRow = 1
        int obCount = array.size(obs)
        if obCount > 0
            for ai = obCount - 1 to 0
                if shownSetups >= Inp5mTableRows or outRow > 20
                    break
                ObZone az = array.get(obs, ai)
                if recentH4ObSelected(ai, h4AuditFrom) and az.triggerTime >= h4AuditFrom and not az.rejected
                    string side = az.bullish ? "BUY" : "SELL"
                    string poi = str.tostring(az.zb, format.mintick) + "–" + str.tostring(az.zt, format.mintick)
                    string impact = az.impactTime >= 0 ? str.format_time(az.impactTime, "MM-dd HH:mm") : "—"
                    string baseStage = az.state == 3 ? "READY FOR 5m" : az.state == 2 ? "H4 OOB / DEAD" : az.eligibilityTime >= 0 ? "WAIT H4 IMPACT" : "WAIT H4"
                    string ledgerStage = ai < array.size(fiveStageByOb) ? array.get(fiveStageByOb, ai) : "—"
                    string setupStage = az.impactTime >= 0 and ledgerStage != "—" ? ledgerStage : baseStage

                    // Count this H4 setup's trades in the deterministic trade ledger.
                    int tradeCount = 0
                    int nt = array.size(fiveTradeOb)
                    if nt > 0
                        for ti = 0 to nt - 1
                            if array.get(fiveTradeOb, ti) == ai
                                tradeCount += 1

                    if tradeCount == 0
                        string result = (setupStage == "H4 OB BREACHED" or setupStage == "H4 CLOSE INVALID") ? "NO TRADE" : az.state == 2 ? "NO TRADE" : az.state == 3 ? "ACTIVE" : "WAIT"
                        table.cell(fiveMinuteEntryTable, 0, outRow, "#" + str.tostring(ai + 1))
                        table.cell(fiveMinuteEntryTable, 1, outRow, side, text_color=az.bullish ? color.blue : color.black)
                        table.cell(fiveMinuteEntryTable, 2, outRow, poi)
                        table.cell(fiveMinuteEntryTable, 3, outRow, impact)
                        table.cell(fiveMinuteEntryTable, 4, outRow, setupStage)
                        table.cell(fiveMinuteEntryTable, 5, outRow, "—")
                        table.cell(fiveMinuteEntryTable, 6, outRow, "—")
                        table.cell(fiveMinuteEntryTable, 7, outRow, "—")
                        table.cell(fiveMinuteEntryTable, 8, outRow, result)
                        table.cell(fiveMinuteEntryTable, 9, outRow, "—")
                        table.cell(fiveMinuteEntryTable, 10, outRow, "—")
                        outRow += 1
                    else
                        // First trade carries parent H4 context. Additional trades get child rows
                        // with blank parent columns so the hierarchy is visually obvious.
                        for attempt = 1 to tradeCount
                            if outRow > 20
                                break
                            int found = -1
                            if nt > 0
                                for ti = 0 to nt - 1
                                    if found == -1 and array.get(fiveTradeOb, ti) == ai and array.get(fiveTradeNo, ti) == attempt
                                        found := ti
                            if found >= 0
                                int entT = array.get(fiveTradeEntryT, found)
                                float entP = array.get(fiveTradeEntryP, found)
                                float slP = array.get(fiveTradeSL, found)
                                float tpP = array.get(fiveTradeTP, found)
                                string trResult = array.get(fiveTradeResult, found)
                                float mfePipsTbl = array.get(fiveTradeMfePips, found)
                                float mfeRTbl = array.get(fiveTradeMfeR, found)
                                float maePipsTbl = array.get(fiveTradeMaePips, found)
                                float maeRTbl = array.get(fiveTradeMaeR, found)
                                string mfeTxt = mfeRTbl == mfeRTbl ? str.tostring(mfeRTbl, "#.00") + "R / " + str.tostring(mfePipsTbl, "#.0") + "p" : "—"
                                string maeTxt = maeRTbl == maeRTbl ? str.tostring(maeRTbl, "#.00") + "R / " + str.tostring(maePipsTbl, "#.0") + "p" : "—"
                                string entryTxt = entT >= 0 and entP == entP ? str.format_time(entT, "MM-dd HH:mm") + " @ " + str.tostring(entP, format.mintick) : "—"
                                float pipSizeTbl = syminfo.type == "forex" ? syminfo.mintick * 10.0 : syminfo.mintick
                                float riskPipsTbl = slP == slP and entP == entP and pipSizeTbl > 0 ? math.abs(entP - slP) / pipSizeTbl : na
                                string slTxt = slP == slP ? str.tostring(slP, format.mintick) + (riskPipsTbl == riskPipsTbl ? " / " + str.tostring(riskPipsTbl, "#.0") + " pips" : "") : "—"
                                string stageTxt = attempt == 1 ? "ENTRY" : "ENTRY" + str.tostring(attempt)
                                bool firstTrade = attempt == 1
                                table.cell(fiveMinuteEntryTable, 0, outRow, firstTrade ? "#" + str.tostring(ai + 1) : "")
                                table.cell(fiveMinuteEntryTable, 1, outRow, firstTrade ? side : "", text_color=az.bullish ? color.blue : color.black)
                                table.cell(fiveMinuteEntryTable, 2, outRow, firstTrade ? poi : "")
                                table.cell(fiveMinuteEntryTable, 3, outRow, firstTrade ? impact : "")
                                table.cell(fiveMinuteEntryTable, 4, outRow, stageTxt)
                                table.cell(fiveMinuteEntryTable, 5, outRow, entryTxt)
                                table.cell(fiveMinuteEntryTable, 6, outRow, slTxt)
                                table.cell(fiveMinuteEntryTable, 7, outRow, tpP == tpP ? str.tostring(tpP, format.mintick) : "—")
                                table.cell(fiveMinuteEntryTable, 8, outRow, trResult)
                                table.cell(fiveMinuteEntryTable, 9, outRow, mfeTxt)
                                table.cell(fiveMinuteEntryTable, 10, outRow, maeTxt)
                                outRow += 1
                        // If this H4 setup later terminated by same-TF close invalidation,
                        // append one child termination row without repeating parent context.
                        if ledgerStage == "H4 CLOSE INVALID" and outRow <= 20
                            table.cell(fiveMinuteEntryTable, 0, outRow, "")
                            table.cell(fiveMinuteEntryTable, 1, outRow, "")
                            table.cell(fiveMinuteEntryTable, 2, outRow, "")
                            table.cell(fiveMinuteEntryTable, 3, outRow, "")
                            table.cell(fiveMinuteEntryTable, 4, outRow, "H4 CLOSE INVALID")
                            table.cell(fiveMinuteEntryTable, 5, outRow, "—")
                            table.cell(fiveMinuteEntryTable, 6, outRow, "—")
                            table.cell(fiveMinuteEntryTable, 7, outRow, "—")
                            table.cell(fiveMinuteEntryTable, 8, outRow, "TERMINATED")
                            table.cell(fiveMinuteEntryTable, 9, outRow, "—")
                            table.cell(fiveMinuteEntryTable, 10, outRow, "—")
                            outRow += 1
                    shownSetups += 1
        if shownSetups == 0
            table.cell(fiveMinuteEntryTable, 0, 1, "No H4 setup in range", text_color=color.gray)
